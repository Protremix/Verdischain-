package com.verdis.verdis_wallet

import dev.sublab.sr25519.*
import dev.sublab.sr25519.secretKey.*
import dev.sublab.sr25519.publicKey.*
import dev.sublab.sr25519.signer.*
import dev.sublab.sr25519.verify.*
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel
import java.util.Base64

/**
 * Native sr25519 key derivation and signing service.
 * Replaces server-side mnemonic exposure with on-device cryptographic operations.
 *
 * Security model:
 * - Mnemonic NEVER leaves the device
 * - Private keys NEVER leave the device
 * - Only public keys and signed payloads are shared
 */
class Sr25519Service {

    companion object {
        private const val SS58_PREFIX: Short = 909

        /**
         * Derive sr25519 keypair from a BIP39 mnemonic seed.
         * Returns: { "address": "xx...", "publicKey": "hex...", "secretKey": "hex..." }
         */
        fun deriveFromMnemonic(mnemonic: String): Map<String, String> {
            // Convert mnemonic to seed using BIP39-BIP32-Ed25519 (Substrate default)
            // Substrate uses mini-secret from the first 32 bytes of Blake2b(mnemonic)
            // For sr25519, the seed is derived from the mnemonic via Blake2b-512
            val seed = mnemonicToSeed(mnemonic)

            // Create mini-secret (first 32 bytes of the seed)
            val miniSecretBytes = seed.copyOfRange(0, 32)

            // Derive sr25519 keypair from mini-secret
            val secretKey = Sr25519SecretKey.fromMiniSecret(miniSecretBytes)
            val publicKey = secretKey.toPublicKey()

            // Encode public key as SS58 address with prefix 909
            val address = publicKey.toSS58Address(SS58_PREFIX)

            return mapOf(
                "address" to address,
                "publicKey" to publicKey.toHex(),
                "secretKey" to secretKey.toHex()
            )
        }

        /**
         * Sign a payload with sr25519.
         * Returns base64-encoded signature.
         */
        fun sign(secretKeyHex: String, payloadHex: String): String {
            val secretKey = Sr25519SecretKey.fromHex(secretKeyHex)
            val payload = hexToBytes(payloadHex)
            val signature = secretKey.sign(payload)
            return bytesToBase64(signature)
        }

        /**
         * Convert BIP39 mnemonic to 64-byte seed using Blake2b-512.
         * This matches Substrate's BIP39 implementation.
         */
        private fun mnemonicToSeed(mnemonic: String): ByteArray {
            // Substrate uses Blake2b-512 on the UTF-8 mnemonic string
            val mnemonicBytes = mnemonic.toByteArray(Charsets.UTF_8)
            return blake2b512(mnemonicBytes)
        }

        /**
         * Blake2b-512 implementation (using Java's MessageDigest equivalent).
         * We use BouncyCastle via the Sublab SDK which includes Blake2b.
         */
        private fun blake2b512(input: ByteArray): ByteArray {
            // Use the Sublab SDK's hashing utilities
            return dev.sublab.hashing.Hashing.blake2b(input, 64)
        }

        private fun hexToBytes(hex: String): ByteArray {
            return hex.chunked(2).map { it.toInt(16).toByte() }.toByteArray()
        }

        private fun bytesToBase64(bytes: ByteArray): String {
            return Base64.getEncoder().encodeToString(bytes)
        }
    }
}

class MainActivity: FlutterActivity() {
    private val CHANNEL = "com.verdis.verdis_wallet/sr25519"

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL).setMethodCallHandler { call, result ->
            try {
                when (call.method) {
                    "deriveAddress" -> {
                        val mnemonic = call.argument<String>("mnemonic") ?: ""
                        val derived = Sr25519Service.deriveFromMnemonic(mnemonic)
                        result.success(derived)
                    }
                    "sign" -> {
                        val secretKeyHex = call.argument<String>("secretKeyHex") ?: ""
                        val payloadHex = call.argument<String>("payloadHex") ?: ""
                        val signature = Sr25519Service.sign(secretKeyHex, payloadHex)
                        result.success(signature)
                    }
                    else -> result.notImplemented()
                }
            } catch (e: Exception) {
                result.error("SR25519_ERROR", e.message, null)
            }
        }
    }
}
