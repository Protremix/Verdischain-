package com.verdis.verdis_wallet

import android.webkit.WebView
import android.webkit.WebViewClient
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel
import java.util.concurrent.CountDownLatch
import java.util.concurrent.TimeUnit

/**
 * MainActivity with native sr25519-equivalent local key derivation.
 *
 * SECURITY MODEL:
 * - Mnemonic NEVER leaves the device
 * - Private keys NEVER leave the device
 * - Only public keys and SS58 addresses are shared externally
 *
 * IMPLEMENTATION:
 * Uses a hidden WebView to execute JavaScript crypto (tweetnacl + Ed25519 + SS58)
 * for local key derivation. No server calls for mnemonic operations.
 */
class MainActivity: FlutterActivity() {
    private val CHANNEL = "com.verdis.verdis_wallet/crypto"
    private var webView: WebView? = null
    private var webViewReady = false

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL).setMethodCallHandler { call, result ->
            when (call.method) {
                "initWebView" -> {
                    initWebView()
                    result.success(true)
                }
                "deriveAddress" -> {
                    val mnemonic = call.argument<String>("mnemonic") ?: ""
                    if (!webViewReady) initWebView()
                    deriveAddress(mnemonic, result)
                }
                "generateMnemonic" -> {
                    if (!webViewReady) initWebView()
                    generateMnemonic(result)
                }
                "validateMnemonic" -> {
                    val mnemonic = call.argument<String>("mnnemonic") ?: ""
                    if (!webViewReady) initWebView()
                    validateMnemonic(mnemonic, result)
                }
                else -> result.notImplemented()
            }
        }
    }

    private fun initWebView() {
        runOnUiThread {
            webView = WebView(this@MainActivity).apply {
                settings.javaScriptEnabled = true
                settings.domStorageEnabled = true
                webViewClient = object : WebViewClient() {
                    override fun onPageFinished(view: WebView?, url: String?) {
                        webViewReady = true
                    }
                }
                // Load the bundled signer HTML from Flutter assets
                val assetManager = assets
                try {
                    // Flutter assets are under "flutter_assets/" prefix
                    val html = assetManager.open("flutter_assets/assets/verdis_signer.html").bufferedReader().use { it.readText() }
                    loadDataWithBaseURL("about:blank", html, "text/html", "UTF-8", null)
                } catch (e: Exception) {
                    // Try alternate path
                    try {
                        val html = assetManager.open("flutter_assets/verdis_signer.html").bufferedReader().use { it.readText() }
                        loadDataWithBaseURL("about:blank", html, "text/html", "UTF-8", null)
                    } catch (e2: Exception) {
                        android.util.Log.e("VerdisSigner", "Failed to load signer HTML: ${e2.message}")
                    }
                }
            }
        }
        // Wait for WebView to be ready
        Thread.sleep(500)
    }

    private fun deriveAddress(mnemonic: String, result: io.flutter.plugin.common.MethodChannel.Result) {
        val latch = CountDownLatch(1)
        var resultJson: String? = null

        runOnUiThread {
            webView?.evaluateJavascript(
                "VerdisSigner.deriveAddress('${mnemonic.replace("'", "\\'")}')",
                { value ->
                    resultJson = value
                    latch.countDown()
                }
            ) ?: run {
                result.error("WEBVIEW_ERROR", "WebView not initialized", null)
                latch.countDown()
            }
        }

        if (latch.await(10, TimeUnit.SECONDS)) {
            // evaluateJavascript returns a JSON string with quotes
            val cleaned = resultJson?.trim()
                ?.removeSurrounding("\"")
                ?.replace("\\\"", "\"")
                ?.replace("\\\\", "\\")
                ?.replace("\\n", "\n")

            if (cleaned != null && cleaned.startsWith("{")) {
                try {
                    val parsed = org.json.JSONObject(cleaned)
                    if (parsed.has("error")) {
                        result.error("DERIVE_ERROR", parsed.getString("error"), null)
                    } else {
                        result.success(mapOf(
                            "address" to parsed.getString("address"),
                            "publicKey" to parsed.getString("publicKey"),
                            "secretKey" to parsed.getString("secretKey")
                        ))
                    }
                } catch (e: Exception) {
                    result.error("PARSE_ERROR", e.message, null)
                }
            } else {
                result.error("DERIVE_ERROR", "Invalid response: $cleaned", null)
            }
        } else {
            result.error("TIMEOUT", "Address derivation timed out", null)
        }
    }

    private fun generateMnemonic(result: io.flutter.plugin.common.MethodChannel.Result) {
        val latch = CountDownLatch(1)
        var resultStr: String? = null

        runOnUiThread {
            webView?.evaluateJavascript(
                "VerdisSigner.generateMnemonic()",
                { value ->
                    resultStr = value
                    latch.countDown()
                }
            ) ?: run {
                result.error("WEBVIEW_ERROR", "WebView not initialized", null)
                latch.countDown()
            }
        }

        if (latch.await(10, TimeUnit.SECONDS)) {
            val cleaned = resultStr?.trim()?.removeSurrounding("\"")
            if (cleaned != null) {
                result.success(cleaned)
            } else {
                result.error("GEN_ERROR", "No mnemonic returned", null)
            }
        } else {
            result.error("TIMEOUT", "Mnemonic generation timed out", null)
        }
    }

    private fun validateMnemonic(mnemonic: String, result: io.flutter.plugin.common.MethodChannel.Result) {
        val latch = CountDownLatch(1)
        var resultStr: String? = null

        runOnUiThread {
            webView?.evaluateJavascript(
                "VerdisSigner.validateMnemonic('${mnemonic.replace("'", "\\'")}')",
                { value ->
                    resultStr = value
                    latch.countDown()
                }
            ) ?: run {
                result.error("WEBVIEW_ERROR", "WebView not initialized", null)
                latch.countDown()
            }
        }

        if (latch.await(10, TimeUnit.SECONDS)) {
            val valid = resultStr?.trim() == "true"
            result.success(valid)
        } else {
            result.error("TIMEOUT", "Validation timed out", null)
        }
    }
}
